cmake_minimum_required(VERSION 3.20)
project(PocketLeagueStoryPort LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(PLS_BUILD_SDL "Build SDL platform fixtures" ON)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc /fp:strict /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/kairo/common/semantic_runtime.cpp")
  add_library(pls_semantics src/kairo/common/semantic_runtime.cpp src/kairo/common/managed_streams.cpp)
  target_include_directories(pls_semantics PUBLIC src/kairo/common src/kairo/api)
endif()
file(GLOB PLS_PORTABLE_GENERATED CONFIGURE_DEPENDS src/kairo/common/generated/*.cpp)
if(PLS_PORTABLE_GENERATED)
  add_library(pls_portable_generated ${PLS_PORTABLE_GENERATED})
  target_include_directories(pls_portable_generated PUBLIC src/kairo/common src/kairo/api src/kairo/common/generated)
  target_link_libraries(pls_portable_generated PUBLIC pls_semantics)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/kairo/common/services.cpp")
  add_library(pls_services src/kairo/common/services.cpp)
  target_include_directories(pls_services PUBLIC src/kairo/common)
endif()
if(PLS_BUILD_SDL)
  find_package(SDL2 CONFIG REQUIRED)
  add_library(pls_sdl src/kairo/platform/sdl/backend.cpp)
  target_include_directories(pls_sdl PUBLIC src/kairo/api src/kairo/platform/sdl)
  target_compile_definitions(pls_sdl PUBLIC SDL_MAIN_HANDLED)
  target_link_libraries(pls_sdl PUBLIC SDL2::SDL2)
  add_executable(sdl_backend_test tests/kairo/sdl_backend_test.cpp)
  target_link_libraries(sdl_backend_test PRIVATE pls_sdl)
  add_executable(asset_roundtrip_test tests/kairo/asset_roundtrip.cpp)
  target_link_libraries(asset_roundtrip_test PRIVATE pls_sdl)
  add_test(NAME sdl_backend_fixture COMMAND sdl_backend_test)
  if(WIN32)
    add_custom_command(TARGET sdl_backend_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:sdl_backend_test>)
    add_custom_command(TARGET asset_roundtrip_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:asset_roundtrip_test>)
  endif()
endif()
