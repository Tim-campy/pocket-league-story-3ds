cmake_minimum_required(VERSION 3.20)
project(PocketLeagueStoryPort LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(WIN32)
  add_executable(pc_metrics_test tests/kairo/pc_metrics_fixtures.cpp src/main/pc_metrics.cpp src/main/pc_memory.cpp)
  target_include_directories(pc_metrics_test PRIVATE src/main)
  target_link_libraries(pc_metrics_test PRIVATE Psapi)
  add_test(NAME pc_metrics_fixture COMMAND pc_metrics_test)
endif()
option(PLS_BUILD_SDL "Build SDL platform fixtures" ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  add_compile_options(/W4 /permissive- /EHsc /fp:strict /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/kairo/common/semantic_runtime.cpp")
  set(PLS_MONO_TABLE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/generated/mono-collation")
  if(NOT EXISTS "${PLS_MONO_TABLE_DIR}/mono_collation_tables.inc" OR NOT EXISTS "${PLS_MONO_TABLE_DIR}/mono_character_tables.inc" OR NOT EXISTS "${PLS_MONO_TABLE_DIR}/mono_decimal_digit_ranges.inc")
    message(FATAL_ERROR "Prepare the authorized official Mono sources with tools/setup_mono_collation.py --apk <your-local-apk>, then run tools/generate_mono_collator.py. No Unity binary is used by this build.")
  endif()
  find_package(Threads REQUIRED)
  add_library(pls_semantics src/kairo/common/semantic_runtime.cpp src/kairo/common/managed_streams.cpp src/kairo/common/managed_exceptions.cpp
    src/kairo/common/managed_encoding.cpp src/kairo/common/managed_regex.cpp src/kairo/common/managed_thread.cpp src/kairo/common/managed_collation.cpp src/kairo/common/managed_mono_character_data.cpp src/kairo/common/managed_stream_reader.cpp src/kairo/common/managed_string_ops.cpp)
  target_include_directories(pls_semantics PUBLIC src/kairo/common src/kairo/api PRIVATE "${PLS_MONO_TABLE_DIR}")
  target_link_libraries(pls_semantics PUBLIC Threads::Threads)
endif()
add_library(pls_assets src/kairo/common/asset_catalog.cpp src/kairo/common/asset_reader.cpp)
target_include_directories(pls_assets PUBLIC src/kairo/common)
target_link_libraries(pls_assets PUBLIC pls_semantics)
add_library(pls_storage src/kairo/common/storage.cpp)
target_include_directories(pls_storage PUBLIC src/kairo/api src/kairo/common)
target_link_libraries(pls_storage PUBLIC pls_assets pls_semantics)
add_library(pls_graphics src/kairo/common/graphics.cpp src/kairo/common/font.cpp src/kairo/common/image_lifetime.cpp)
target_include_directories(pls_graphics PUBLIC src/kairo/api src/kairo/common)
target_link_libraries(pls_graphics PUBLIC pls_assets pls_semantics)
add_executable(graphics_api_test tests/kairo/graphics_api_test.cpp)
target_link_libraries(graphics_api_test PRIVATE pls_graphics)
add_executable(asset_catalog_test tests/kairo/asset_catalog_test.cpp)
target_link_libraries(asset_catalog_test PRIVATE pls_assets)
file(GLOB PLS_PORTABLE_GENERATED CONFIGURE_DEPENDS src/kairo/common/generated/*.cpp)
if(PLS_PORTABLE_GENERATED)
  add_library(pls_portable_generated ${PLS_PORTABLE_GENERATED})
  target_include_directories(pls_portable_generated PUBLIC src/kairo/common src/kairo/api src/kairo/common/generated)
  target_link_libraries(pls_portable_generated PUBLIC pls_semantics)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/kairo/common/services.cpp")
  add_library(pls_services src/kairo/common/services.cpp)
  target_include_directories(pls_services PUBLIC src/kairo/common)
endif()
if(PLS_BUILD_SDL)
  find_package(SDL2 CONFIG REQUIRED)
  add_library(pls_sdl src/kairo/platform/sdl/backend.cpp src/kairo/platform/sdl/globalization.cpp src/kairo/platform/sdl/thread_priority.cpp src/kairo/platform/sdl/debug_input.cpp)
  target_include_directories(pls_sdl PUBLIC src/kairo/api src/kairo/platform/sdl)
  target_compile_definitions(pls_sdl PUBLIC SDL_MAIN_HANDLED)
  target_link_libraries(pls_sdl PUBLIC SDL2::SDL2 pls_semantics)
  if(WIN32)
    target_link_libraries(pls_sdl PUBLIC Advapi32)
  endif()
  if(TARGET pls_portable_generated)
    target_link_libraries(pls_portable_generated PUBLIC pls_sdl)
  endif()
  list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/native-media")
  find_package(freetype CONFIG REQUIRED)
  find_package(Ogg CONFIG REQUIRED)
  find_package(Vorbis CONFIG REQUIRED)
  add_library(pls_media src/kairo/platform/sdl/font.cpp src/kairo/platform/sdl/audio.cpp)
  target_include_directories(pls_media PUBLIC src/kairo/platform/sdl)
  target_link_libraries(pls_media PUBLIC SDL2::SDL2 Freetype::Freetype Vorbis::vorbisfile)
  target_link_libraries(pls_graphics PUBLIC pls_media)
  add_library(pls_ui src/kairo/platform/sdl/input.cpp src/kairo/common/canvas.cpp src/kairo/common/forms.cpp src/kairo/common/forms_execution.cpp
    src/kairo/common/dialog_service.cpp src/kairo/platform/sdl/dialog.cpp src/kairo/common/bootstrap.cpp)
  target_include_directories(pls_ui PUBLIC src/kairo/api src/kairo/common src/kairo/platform/sdl)
  target_link_libraries(pls_ui PUBLIC pls_graphics pls_sdl)
  add_library(pls_sound src/kairo/common/sound.cpp)
  target_include_directories(pls_sound PUBLIC src/kairo/api src/kairo/common)
  target_link_libraries(pls_sound PUBLIC pls_media pls_assets pls_semantics)
  target_link_libraries(pls_ui PUBLIC pls_sound)
  set(PLS_DIAGNOSTIC_GAME_DIR "" CACHE PATH "Private diagnostic generated sources; never a normal game build")
  set(PLS_GAME_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/game")
  if(PLS_DIAGNOSTIC_GAME_DIR)
    if(NOT EXISTS "${PLS_DIAGNOSTIC_GAME_DIR}/manifest.json")
      message(FATAL_ERROR "A diagnostic manifest verified by tools/build-pc is required")
    endif()
    set(PLS_GAME_INCLUDE_DIR "${PLS_DIAGNOSTIC_GAME_DIR}")
  endif()
  file(GLOB_RECURSE PLS_GAME_GENERATED CONFIGURE_DEPENDS "${PLS_GAME_INCLUDE_DIR}/*.cpp")
  add_executable(pc_game EXCLUDE_FROM_ALL src/main/pc_main.cpp src/main/pc_memory.cpp src/main/pc_metrics.cpp ${PLS_GAME_GENERATED})
  if(MSVC)
    # Generated C# bodies exceed the standard COFF section limit in Debug.
    target_compile_options(pc_game PRIVATE /bigobj)
  endif()
  target_include_directories(pc_game PRIVATE src/kairo/api "${PLS_GAME_INCLUDE_DIR}")
  if(PLS_DIAGNOSTIC_GAME_DIR)
    target_compile_definitions(pc_game PRIVATE PLS_DIAGNOSTIC_BUILD=1)
  endif()
  target_link_libraries(pc_game PRIVATE pls_semantics pls_portable_generated pls_graphics pls_services pls_storage pls_media pls_sdl pls_ui pls_sound)
  if(WIN32)
    target_link_libraries(pc_game PRIVATE Psapi)
    add_custom_command(TARGET pc_game POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:pc_game>)
  endif()
  add_executable(sdl_backend_test tests/kairo/sdl_backend_test.cpp)
  target_link_libraries(sdl_backend_test PRIVATE pls_sdl)
  add_executable(debug_input_test tests/kairo/debug_input_fixtures.cpp)
  target_link_libraries(debug_input_test PRIVATE pls_sdl)
  add_test(NAME debug_input_fixture COMMAND debug_input_test policy)
  add_executable(asset_roundtrip_test tests/kairo/asset_roundtrip.cpp)
  target_link_libraries(asset_roundtrip_test PRIVATE pls_sdl)
  add_test(NAME sdl_backend_fixture COMMAND sdl_backend_test)
  if(WIN32)
    add_custom_command(TARGET sdl_backend_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:sdl_backend_test>)
    add_custom_command(TARGET debug_input_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:debug_input_test>)
    add_custom_command(TARGET asset_roundtrip_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:asset_roundtrip_test>)
  endif()
endif()
