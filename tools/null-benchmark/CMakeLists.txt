cmake_minimum_required(VERSION 3.20)
project(PlsNullBenchmark LANGUAGES C CXX)
if(MSVC AND DEFINED PLS_MSVC_SHOWINCLUDES_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${PLS_MSVC_SHOWINCLUDES_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT PLS_GENERATED OR NOT EXISTS "${PLS_GENERATED}/manifest.json")
  message(FATAL_ERROR "Use a manifest-verified private snapshot; missing bodies stay fatal traps")
endif()
add_compile_definitions(PLS_NULL_BACKEND=1)
option(PLS_TRACING_GC "Use the nonmoving tracing runtime" ON)
if(PLS_TRACING_GC)
  add_compile_definitions(PLS_TRACING_GC=1)
else()
  add_compile_definitions(PLS_TRACING_GC=0)
endif()
option(PLS_ALLOCATION_PROBE "Separate allocation run; never a CPU timing claim" OFF)
option(PLS_REPORT_ALLOCATION_FAILURE "ARM startup failure backtrace; not a timing build" OFF)
option(PLS_MEMORY_CATEGORIES "Separate known-range memory diagnosis; not CPU measurement" OFF)
option(PLS_GC_TRACE "One stderr line per collection; diagnosis only, not a timing build" OFF)
option(PLS_GC_LEGACY_RENDEZVOUS "Diagnosis only: CTR GC rendezvous uses the former timed condvar wait" OFF)
if(PLS_GC_LEGACY_RENDEZVOUS)
  add_compile_definitions(PLS_GC_LEGACY_RENDEZVOUS=1)
endif()
if(PLS_GC_TRACE)
  add_compile_definitions(PLS_GC_TRACE=1)
endif()
if(PLS_MEMORY_CATEGORIES)
  add_compile_definitions(PLS_MEMORY_CATEGORIES=1)
endif()
set(PLS_ARM_OPTIMIZATION "O2" CACHE STRING "ARM compiler optimization: O2 or Os")
set_property(CACHE PLS_ARM_OPTIMIZATION PROPERTY STRINGS O2 Os)
option(PLS_ARM_LTO "ARM link-time optimization measurement" OFF)
option(PLS_ARM_GC_SECTIONS "Discard unused ARM sections" ON)
if(PLS_ALLOCATION_PROBE)
  add_compile_definitions(PLS_ALLOCATION_PROBE=1)
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  add_compile_options(/O2 /Zi /bigobj /EHsc /fp:strict /permissive- /utf-8 /we4715)
  add_link_options(/DEBUG /OPT:REF /OPT:ICF)
else()
  if(NOT PLS_ARM_OPTIMIZATION MATCHES "^(O2|Os)$")
    message(FATAL_ERROR "Unsupported ARM optimization measurement")
  endif()
  add_compile_options(-${PLS_ARM_OPTIMIZATION} -g -fexceptions -frtti -fno-fast-math -ffp-contract=off -ffunction-sections -fdata-sections)
  if(PLS_ARM_GC_SECTIONS)
    add_link_options(-Wl,--gc-sections)
  else()
    # The SDK specs also enable GC. An absent option would not disable it.
    add_link_options(-Wl,--no-gc-sections)
  endif()
  if(PLS_ARM_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT PLS_IPO_SUPPORTED OUTPUT PLS_IPO_ERROR)
    if(NOT PLS_IPO_SUPPORTED)
      message(FATAL_ERROR "LTO unavailable: ${PLS_IPO_ERROR}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
find_package(freetype CONFIG REQUIRED)
set(COMMON semantic_runtime managed_streams managed_exceptions managed_encoding managed_regex managed_thread managed_collation managed_mono_character_data managed_stream_reader managed_string_ops managed_float_format asset_catalog asset_reader storage graphics mesh font image_lifetime services canvas forms forms_execution dialog_service bootstrap text_input offscreen sound)
set(SOURCES)
foreach(name IN LISTS COMMON)
  list(APPEND SOURCES "${ROOT}/src/kairo/common/${name}.cpp")
endforeach()
list(APPEND SOURCES "${ROOT}/src/kairo/platform/sdl/font.cpp" "${ROOT}/src/kairo/platform/sdl/input.cpp" "${ROOT}/src/kairo/platform/sdl/capture.cpp" "${ROOT}/src/kairo/platform/sdl/globalization.cpp" "${ROOT}/src/kairo/platform/null/platform.cpp")
if(WIN32)
  list(APPEND SOURCES "${ROOT}/src/kairo/platform/sdl/runtime_memory.cpp" "${ROOT}/src/kairo/platform/sdl/thread_priority.cpp")
endif()
file(GLOB PORTABLE "${ROOT}/src/kairo/common/generated/*.cpp")
file(GLOB GAME "${PLS_GENERATED}/*.cpp")
add_library(null_runtime STATIC ${SOURCES} ${PORTABLE})
target_include_directories(null_runtime PUBLIC "${ROOT}/src" "${ROOT}/src/main" "${ROOT}/src/kairo/api" "${ROOT}/src/kairo/common" "${ROOT}/src/kairo/common/generated" "${ROOT}/src/kairo/platform/sdl" PRIVATE "${ROOT}/build/generated/mono-collation" "${ROOT}/build/generated/mono-numberformat")
target_link_libraries(null_runtime PUBLIC Freetype::Freetype)
add_executable(null_game "${ROOT}/src/main/null_main.cpp" ${GAME})
if(PLS_ALLOCATION_PROBE OR PLS_REPORT_ALLOCATION_FAILURE)
  target_sources(null_game PRIVATE "${ROOT}/src/main/allocation_probe.cpp")
endif()
if(PLS_REPORT_ALLOCATION_FAILURE)
  set_source_files_properties("${ROOT}/src/main/allocation_probe.cpp" PROPERTIES COMPILE_DEFINITIONS PLS_REPORT_ALLOCATION_FAILURE=1)
endif()
target_include_directories(null_game PRIVATE "${PLS_GENERATED}")
target_link_libraries(null_game PRIVATE null_runtime)
if(NINTENDO_3DS)
  target_link_options(null_game PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/null_game.map")
endif()
add_executable(null_backend_test "${ROOT}/tests/kairo/null_backend_fixtures.cpp")
add_executable(platform_wait_test "${ROOT}/tests/kairo/platform_wait.cpp")
target_include_directories(platform_wait_test PRIVATE "${ROOT}/src/kairo/common")
target_link_libraries(null_backend_test PRIVATE null_runtime)
if(WIN32)
  target_link_libraries(null_game PRIVATE Psapi Advapi32)
elseif(NINTENDO_3DS)
  include("${ROOT}/tools/cmake/ctr_thread_lifecycle.cmake")
  foreach(target null_game null_backend_test platform_wait_test)
    pls_ctr_thread_lifecycle(${target})
  endforeach()
  ctr_create_3dsx(platform_wait_test NOSMDH)
  ctr_create_3dsx(null_game NOSMDH)
  add_executable(arm11_runtime_smoke "${ROOT}/tests/kairo/arm11_runtime_smoke.cpp")
  target_link_libraries(arm11_runtime_smoke PRIVATE null_runtime)
  pls_ctr_thread_lifecycle(arm11_runtime_smoke)
  ctr_create_3dsx(arm11_runtime_smoke NOSMDH)
endif()
